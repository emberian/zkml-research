314: pub fn karatsuba_add_external_product_assign<Scalar>(
315:     mut out: GlweCiphertextMutView<'_, Scalar>,
316:     ggsw: GgswCiphertextView<Scalar>,
317:     glwe: GlweCiphertextView<Scalar>,
318:     stack: &mut PodStack,
319: ) where
320:     Scalar: UnsignedTorus,
321: {
322:     // we check that the polynomial sizes match
323:     debug_assert_eq!(ggsw.polynomial_size(), glwe.polynomial_size());
324:     debug_assert_eq!(ggsw.polynomial_size(), out.polynomial_size());
325:     // we check that the glwe sizes match
326:     debug_assert_eq!(ggsw.glwe_size(), glwe.glwe_size());
327:     debug_assert_eq!(ggsw.glwe_size(), out.glwe_size());
328: 
329:     let align = CACHELINE_ALIGN;
330:     let poly_size = ggsw.polynomial_size().0;
331: 
332:     // we round the input mask and body
333:     let decomposer = SignedDecomposer::<Scalar>::new(
334:         ggsw.decomposition_base_log(),
335:         ggsw.decomposition_level_count(),
336:     );
337: 
338:     let (output_buffer, substack0) =
339:         stack.make_aligned_raw::<Scalar>(poly_size * ggsw.glwe_size().0, align);
340:     // output_fft_buffer is initially uninitialized, considered to be implicitly zero, to avoid
341:     // the cost of filling it up with zeros. `is_output_uninit` is set to `false` once
342:     // it has been fully initialized for the first time.
343:     let output_buffer = &mut *output_buffer;
344:     let mut is_output_uninit = true;
345: 
346:     let (mut decomposition, substack1) = TensorSignedDecompositionLendingIter::new(
347:         glwe.as_ref()
348:             .iter()
349:             .map(|s| decomposer.init_decomposer_state(*s)),
350:         DecompositionBaseLog(decomposer.base_log),
351:         DecompositionLevelCount(decomposer.level_count),
352:         substack0,
353:     );
354: 
355:     // We loop through the levels
356:     for ggsw_decomp_matrix in ggsw.iter() {
357:         // We retrieve the decomposition of this level.
358:         let (_glwe_level, glwe_decomp_term, _substack2) =
359:             collect_next_term(&mut decomposition, substack1, align);
360:         let glwe_decomp_term = GlweCiphertextView::from_container(
361:             &*glwe_decomp_term,
362:             ggsw.polynomial_size(),
363:             out.ciphertext_modulus(),
364:         );
365: 
366:         // For each level we have to add the result of the vector-matrix product between the
367:         // decomposition of the glwe, and the ggsw level matrix to the output. To do so, we
368:         // iteratively add to the output, the product between every line of the matrix, and
369:         // the corresponding (scalar) polynomial in the glwe decomposition:
370:         //
371:         //                ggsw_mat                        ggsw_mat
372:         //   glwe_dec   | - - - - | <        glwe_dec   | - - - - |
373:         //  | - - - | x | - - - - |         | - - - | x | - - - - | <
374:         //    ^         | - - - - |             ^       | - - - - |
375:         //
376:         //        t = 1                           t = 2                     ...
377: 
378:         for (ggsw_row, glwe_poly) in izip_eq!(
379:             ggsw_decomp_matrix.as_glwe_list().iter(),
380:             glwe_decomp_term.as_polynomial_list().iter()
381:         ) {
382:             let row_as_poly_list = ggsw_row.as_polynomial_list();
383:             if is_output_uninit {
384:                 for (mut output_poly, row_poly) in output_buffer
385:                     .chunks_exact_mut(poly_size)
386:                     .map(Polynomial::from_container)
387:                     .zip(row_as_poly_list.iter())
388:                 {
389:                     polynomial_wrapping_mul(&mut output_poly, &row_poly, &glwe_poly);
390:                 }
391:             } else {
392:                 for (mut output_poly, row_poly) in output_buffer
393:                     .chunks_exact_mut(poly_size)
394:                     .map(Polynomial::from_container)
395:                     .zip(row_as_poly_list.iter())
396:                 {
397:                     polynomial_wrapping_add_mul_assign(&mut output_poly, &row_poly, &glwe_poly);
398:                 }
399:             }
400: 
401:             is_output_uninit = false;
402:         }
403:     }
404: 
405:     // We iterate over the polynomials in the output.
406:     if !is_output_uninit {
407:         izip_eq!(
408:             out.as_mut_polynomial_list().iter_mut(),
409:             output_buffer
410:                 .into_chunks(poly_size)
411:                 .map(Polynomial::from_container),
412:         )
413:         .for_each(|(mut out, res)| polynomial_wrapping_add_assign(&mut out, &res));
414:     }

1: tfhe-ntt is a pure Rust high performance Number Theoretic Transform library that processes
2: vectors of sizes that are powers of two.
3: 
4: This library provides three kinds of NTT:
5: - The prime NTT computes the transform in a field $\mathbb{Z}/p \mathbb{Z}$
6:   with $p$ prime, allowing for arithmetic operations on the polynomial modulo $p$.
7: - The native NTT internally computes the transform of the first kind with
8:   several primes, allowing the simulation of arithmetic modulo the product of
9:   those primes, and truncates the
10: result when the inverse transform is desired. The truncated result is guaranteed to be as if
11: the computations were performed with wrapping arithmetic, as long as the full integer result
12: would have be smaller than half the product of the primes, in absolute value. It is guaranteed
13: to be suitable for multiplying two polynomials with arbitrary coefficients, and returns the
14: result in wrapping arithmetic.
15: - The native binary NTT is similar to the native NTT, but is optimized for the case where one
16: of the operands of the multiplication has coefficients in $\lbrace 0, 1 \rbrace$.
17: 
18: # Rust requirements
19: tfhe-ntt requires a Rust version >= 1.67.0.
20: 
21: # Features
22: 
23: - `std` (default): This enables runtime arch detection for accelerated SIMD instructions.
24: - `avx512` (default): This enables AVX512 instructions on CPUs that support them to further
25:   speed up the NTT.
54: - `mul_poly_prime.rs`: Negacyclic polynomial multiplication with a prime modulus.
55: Run the example with `cargo run --example mul_poly_prime`.
56: - `mul_poly_native.rs`: Negacyclic polynomial multiplication with a native modulus (`2^32`, `2^64`, or `2^128`).
57: Run the example with `cargo run --example mul_poly_native`.
